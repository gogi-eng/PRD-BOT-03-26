import requests
import json
from typing import Dict, Any

class GemmaAI:
    def __init__(self, model="gemma:7b", host="http://localhost:11434"):
        self.model = model
        self.url = f"{host}/api/generate"

    def analyze_trade(self, data: Dict[str, Any]) -> Dict[str, Any]:
        prompt = self._build_prompt(data)

        try:
            response = requests.post(self.url, json={
                "model": self.model,
                "prompt": prompt,
                "stream": False
            }, timeout=10)

            result = response.json()["response"]

            return self._parse_response(result)

        except Exception as e:
            return {
                "approve": True,   # fail-open (как у тебя уже есть логика)
                "confidence": 50,
                "reason": f"Gemma error: {e}"
            }

    def _build_prompt(self, data: Dict[str, Any]) -> str:
        return f"""
You are a professional crypto scalping AI.

Analyze trade:

Symbol: {data.get("symbol")}
Direction: {data.get("side")}
Confidence: {data.get("confidence")}
Trend: {data.get("trend")}
ATR: {data.get("atr")}
ADX: {data.get("adx")}
Volume: {data.get("volume")}
Imbalance: {data.get("imbalance")}

Rules:
- Reject weak or late entries
- Avoid chop zones
- Prefer strong momentum

Return JSON:
{{
 "approve": true/false,
 "confidence": 0-100,
 "reason": "short explanation"
}}
"""
    def _parse_response(self, text: str) -> Dict[str, Any]:
        try:
            return json.loads(text)
        except:
            return {
                "approve": True,
                "confidence": 50,
                "reason": "parse_error"
            }

import numpy as np
from collections import deque

class MetaController:
    def __init__(self):
        self.pnl_history = deque(maxlen=50)
        self.winrate_history = deque(maxlen=50)
        self.drawdown = 0

        self.mode = "NORMAL"
        self.market_regime = "UNKNOWN"

    # =========================
    # MARKET REGIME
    # =========================
    def detect_regime(self, df):
        volatility = df['close'].pct_change().std()
        trend = abs(df['close'].pct_change().mean())

        if volatility > 0.02:
            self.market_regime = "CHAOS"
        elif trend > 0.001:
            self.market_regime = "TREND"
        else:
            self.market_regime = "RANGE"

        return self.market_regime

    # =========================
    # PERFORMANCE TRACKING
    # =========================
    def update_performance(self, pnl):
        self.pnl_history.append(pnl)

        wins = [x for x in self.pnl_history if x > 0]
        winrate = len(wins) / len(self.pnl_history) if self.pnl_history else 0

        self.winrate_history.append(winrate)

        total = sum(self.pnl_history)
        peak = max([sum(list(self.pnl_history)[:i+1]) for i in range(len(self.pnl_history))], default=0)

        self.drawdown = peak - total

    # =========================
    # MODE DECISION
    # =========================
    def decide_mode(self):
        avg_winrate = np.mean(self.winrate_history) if self.winrate_history else 0

        if self.drawdown > 0.05:
            self.mode = "SAFE"

        elif avg_winrate > 0.6:
            self.mode = "AGGRESSIVE"

        else:
            self.mode = "NORMAL"

        return self.mode

    # =========================
    # RISK MULTIPLIER
    # =========================
    def get_risk_multiplier(self):
        if self.mode == "SAFE":
            return 0.5
        elif self.mode == "AGGRESSIVE":
            return 1.5
        return 1.0

    # =========================
    # SHOULD TRADE?
    # =========================
    def allow_trade(self):
        if self.drawdown > 0.1:
            return False

        if self.market_regime == "CHAOS":
            return False

        return True

🔹 DQN (Deep Q Learning)
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import random

# =========================
# NEURAL NETWORK
# =========================
class DQN(nn.Module):
    def __init__(self, state_size, action_size):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_size, 64),
            nn.ReLU(),
            nn.Linear(64, 64),
            nn.ReLU(),
            nn.Linear(64, action_size)
        )

    def forward(self, x):
        return self.net(x)


# =========================
# RL CONTROLLER
# =========================
class RLMetaController:
    def __init__(self):
        self.state_size = 7
        self.action_size = 4

        self.model = DQN(self.state_size, self.action_size)
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)

        self.memory = []
        self.gamma = 0.95

        self.epsilon = 1.0
        self.epsilon_decay = 0.995
        self.epsilon_min = 0.05

    # =========================
    # ACTION
    # =========================
    def act(self, state):
        if np.random.rand() < self.epsilon:
            return random.randint(0, self.action_size - 1)

        state = torch.FloatTensor(state)
        q_values = self.model(state)

        return torch.argmax(q_values).item()

    # =========================
    # STORE EXPERIENCE
    # =========================
    def remember(self, state, action, reward, next_state):
        self.memory.append((state, action, reward, next_state))

        if len(self.memory) > 5000:
            self.memory.pop(0)

    # =========================
    # TRAIN
    # =========================
    def train(self, batch_size=32):
        if len(self.memory) < batch_size:
            return

        batch = random.sample(self.memory, batch_size)

        for state, action, reward, next_state in batch:
            target = reward

            next_q = self.model(torch.FloatTensor(next_state))
            target += self.gamma * torch.max(next_q).item()

            q_values = self.model(torch.FloatTensor(state))
            target_vec = q_values.clone().detach()
            target_vec[action] = target

            loss = nn.MSELoss()(q_values, target_vec)

            self.optimizer.zero_grad()
            loss.backward()
            self.optimizer.step()

        self.epsilon = max(self.epsilon_min, self.epsilon * self.epsilon_decay)

import numpy as np

class HybridVoter:
    def __init__(self):
        # стартовые веса
        self.weights = {
            "xgb": 0.4,
            "gemma": 0.3,
            "ta": 0.3
        }

        # адаптивная статистика
        self.performance = {
            "xgb": [],
            "gemma": [],
            "ta": []
        }

        self.threshold = 0.2  # минимальный сигнал

    # =========================
    # НОРМАЛИЗАЦИЯ
    # =========================
    def normalize(self, signal):
        return max(-1, min(1, signal))

    # =========================
    # ВЕСОВОЕ ГОЛОСОВАНИЕ
    # =========================
    def vote(self, signals):
        """
        signals = {
            "xgb": 0.7,
            "gemma": 0.5,
            "ta": -0.2
        }
        """

        total = 0
        weight_sum = 0

        for key, value in signals.items():
            value = self.normalize(value)
            w = self.weights.get(key, 0)

            total += value * w
            weight_sum += w

        if weight_sum == 0:
            return 0

        score = total / weight_sum
        return score

    # =========================
    # DECISION
    # =========================
    def decide(self, score):
        if score > self.threshold:
            return "LONG"
        elif score < -self.threshold:
            return "SHORT"
        return "NO_TRADE"

    # =========================
    # ОБНОВЛЕНИЕ ВЕСОВ
    # =========================
    def update_performance(self, results):
        """
        results = {
            "xgb": pnl,
            "gemma": pnl,
            "ta": pnl
        }
        """

        for k, v in results.items():
            self.performance[k].append(v)

        self._recalculate_weights()

    def _recalculate_weights(self):
        scores = {}

        for k, history in self.performance.items():
            if len(history) == 0:
                scores[k] = 0
            else:
                scores[k] = np.mean(history)

        total = sum([abs(v) for v in scores.values()]) + 1e-8

        for k in self.weights:
            self.weights[k] = abs(scores[k]) / total
